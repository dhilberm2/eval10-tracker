function spark(vals,color){
 const pts=vals.map((v,i)=>({v,i})).filter(p=>p.v!=null);
 if(pts.length<2)return '<span class="mute">needs 2+ days</span>';
 const mn=Math.min(...pts.map(p=>p.v)),mx=Math.max(...pts.map(p=>p.v)),rg=mx-mn||1;
 const path=pts.map((p,k)=>(k?"L":"M")+(p.i/9*280+10).toFixed(1)+","+(40-(p.v-mn)/rg*32).toFixed(1)).join(" ");
 return '<svg viewBox="0 0 300 46" preserveAspectRatio="none"><path d="'+path+'" fill="none" stroke="'+color+'" stroke-width="2"/></svg>';
}

const f=(v,n=1)=>v==null?"–":v.toFixed(n);

(~"+e2.toFixed(0)+" kg 1RM)":"");C=a.length>1&&fi.w===la.w?(la.r-fi.r>=0?"+":"")+(la.r-fi.r)+" reps at same load":(e1&&e2&&a.length>1?((e2/e1-1)*100).toFixed(1)+"% est. 1RM":"–")}
   else if(ex.startsWith("Cooper")){F=fi.w+" m";L=la.w+" m";C=a.length>1?(la.w-fi.w)+" m":"–"}
   else{F=fi.r+" reps";L=la.r+" reps";C=a.length>1?(la.r-fi.r>=0?"+":"")+(la.r-fi.r)+" reps":"–"}
   h+="<tr><td>"+esc(ex)+"</td><td>"+F+"</td><td>"+L+"</td><td>"+C+"</td></tr>"});
  h+="</table></div>"}
